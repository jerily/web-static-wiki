cmake_minimum_required(VERSION 3.18)
project(web-static-wiki VERSION 0.1.0 LANGUAGES NONE)
set(PACKAGE_DIR "lib/web-static-wiki${PROJECT_VERSION}")
install(FILES pkgIndex.tcl readme.md LICENSE DESTINATION "${PACKAGE_DIR}")
install(DIRECTORY library assets DESTINATION "${PACKAGE_DIR}")
install(FILES MOUNTING.md DESTINATION "${PACKAGE_DIR}")
include(CTest)
if(BUILD_TESTING)
    find_program(TCLSH NAMES tclsh9.1 tclsh9.0 tclsh8.6 tclsh REQUIRED)
    add_test(NAME web-static-wiki COMMAND "${TCLSH}" "${CMAKE_CURRENT_SOURCE_DIR}/tests/all.tcl")
endif()
